// Obtém todos os dados estruturados para o app de uma única vez
function obterDadosCompletos() {
  try {
    inicializarPlanilha();
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Identifica o colaborador logado e sua agência (controle de acesso)
    var usuario = obterUsuarioAtual();

    // 1. Carrega Associados (leitura orientada por nome de coluna)
    var shAssociados = ss.getSheetByName("Associados");
    var mapaA = garantirColunasAssociados();
    var dadosAssociados = shAssociados.getDataRange().getValues();
    var associados = [];
    for (var i = 1; i < dadosAssociados.length; i++) {
      var r = dadosAssociados[i];
      var g = function(nome) { var idx = mapaA[nome]; return (idx === undefined) ? "" : r[idx]; };
      associados.push({
        id: g("id"),
        nome: g("nome"),
        cpfCnpj: g("cpf/cnpj"),
        conta: g("conta"),
        limiteAtual: Number(g("limite atual")) || 0,
        sugestaoMotor: Number(g("sugestao motor")) || 0,
        status: g("status") || "AGUARDANDO",
        rendaComprovada: Number(g("renda comprovada")) || 0,
        comprometimentoSCR: Number(g("comprometimento scr")) || 0,
        score: Number(g("score")) || 0,
        inadimplenciaDias: Number(g("inadimplencia dias")) || 0,
        tempoAssociadoAnos: Number(g("tempo associado (anos)")) || 0,
        justificativa: g("justificativa") || "",
        novoLimiteConfirmado: g("novo limite confirmado") !== "" ? Number(g("novo limite confirmado")) : null,
        agencia: String(g("agencia") || "").trim(),
        microSegmento: String(g("micro segmento") || "").trim(),
        ativoComercial: String(g("ativo comercial") || "").trim(),
        irc: String(g("irc") || "").trim(),
        capitalSocial: Number(g("capital social")) || 0,
        dtBaseScr: String(g("dt base scr") || "").trim(),
        emprestScr: Number(g("emprest. e financ. scr")) || 0,
        emprest360: Number(g("emprest. e financ. 360 scr")) || 0,
        emprestMensal: Number(g("emprest. e financ. mensal scr")) || 0,
        limiteScr: Number(g("limite scr")) || 0,
        saldoDevedor: Number(g("saldo devedor")) || 0
      });
    }

    // Aplica o filtro por agência:
    // - SEDE: vê todos os associados
    // - agência específica: vê apenas os da sua agência
    // - não cadastrado: não vê nenhum associado
    if (!usuario.isSede) {
      if (usuario.cadastrado) {
        var agUsuario = normalizarCol(usuario.agencia);
        associados = associados.filter(function(a) {
          return normalizarCol(a.agencia) === agUsuario;
        });
      } else {
        associados = [];
      }
    }

    // 2. Carrega Uploads
    var shUploads = ss.getSheetByName("Uploads");
    var dadosUploads = shUploads.getDataRange().getValues();
    var uploads = [];
    for (var i = 1; i < dadosUploads.length; i++) {
      var r = dadosUploads[i];
      uploads.push({
        id: r[0],
        arquivo: r[1],
        tipoLista: r[2],
        dataHora: r[3],
        registros: Number(r[4]) || 0,
        status: r[5] || "Validado",
        detalhesErro: r[6] || ""
      });
    }

    // 3. Carrega Protocolos
    var shProtocolos = ss.getSheetByName("Protocolos");
    var dadosProtocolos = shProtocolos.getDataRange().getValues();
    var protocolos = [];
    for (var i = 1; i < dadosProtocolos.length; i++) {
      var r = dadosProtocolos[i];
      protocolos.push({
        id: r[0],
        associado: r[1],
        tipo: r[2],
        valorSolicitado: Number(r[3]) || 0,
        data: r[4],
        status: r[5] || "Pendente"
      });
    }

    // Calcular estatísticas financeiras gerais para a agência
    var totalPendente = 0;
    protocolos.forEach(function(p) {
      if (p.status === "Pendente") {
        totalPendente += p.valorSolicitado;
      }
    });

    var limiteContaTotal = 0;
    associados.forEach(function(a) {
      limiteContaTotal += a.limiteAtual;
    });

    return {
      usuario: usuario,
      associados: associados,
      uploads: uploads.reverse(), // Ordena mais novos primeiro
      protocolos: protocolos,
      estatisticas: {
        totalPendente: totalPendente || 1245000,
        limiteContaTotal: limiteContaTotal || 8920450,
        limiteCartaoTotal: 3412000,
        limiteContaUtilizadoPercent: 92,
        revisaoDiasRestantes: 2
      }
    };
  } catch (err) {
    return { error: err.toString() };
  }
}

// Atualiza o limite de crédito do associado e muda seu status
function confirmarNovoLimite(id, novoLimite, justificativa) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Associados");
    var m = mapaColunas(sheet);
    var data = sheet.getDataRange().getValues();

    var colId = m["id"], colNome = m["nome"], colStatus = m["status"];
    var colJust = m["justificativa"], colNovo = m["novo limite confirmado"];

    for (var i = 1; i < data.length; i++) {
      if (data[i][colId] === id) {
        var linha = i + 1;
        sheet.getRange(linha, colStatus + 1).setValue("APROVADO");
        sheet.getRange(linha, colJust + 1).setValue(justificativa);
        sheet.getRange(linha, colNovo + 1).setValue(novoLimite);

        // Adiciona um protocolo de aprovação registrado
        var shProtocolos = ss.getSheetByName("Protocolos");
        var novoId = "#2024-" + String(shProtocolos.getLastRow()).padStart(3, '0');
        shProtocolos.appendRow([novoId, data[i][colNome], "Limite de Conta", novoLimite, obterDataAtualFormatada(), "Aprovado"]);

        return { success: true };
      }
    }
    return { success: false, error: "Associado não encontrado" };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

// Cria um Novo Protocolo
function registrarNovoProtocolo(associado, tipo, valorSolicitado) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Protocolos");
    var novoId = "#2024-" + String(sheet.getLastRow()).padStart(3, '0');
    var dataStr = obterDataAtualFormatada();
    sheet.appendRow([novoId, associado, tipo, valorSolicitado, dataStr, "Pendente"]);
    return { success: true, protocolo: { id: novoId, associado: associado, tipo: tipo, valorSolicitado: valorSolicitado, data: dataStr, status: "Pendente" } };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

// Registra um novo upload no histórico
function registrarUpload(arquivo, tipoLista, registros, status, detalhesErro) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Uploads");
    var novoId = "UP-" + String(sheet.getLastRow()).padStart(3, '0');
    var dataHoraStr = obterDataAtualFormatada() + " - " + obterHoraAtualFormatada();
    sheet.appendRow([novoId, arquivo, tipoLista, dataHoraStr, registros, status, detalhesErro || ""]);
    return { success: true, upload: { id: novoId, arquivo: arquivo, tipoLista: tipoLista, dataHora: dataHoraStr, registros: registros, status: status, detalhesErro: detalhesErro } };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

// Valida e processa um arquivo real de texto (CSV/TSV) contendo as bases de majoração
function validarEProcessarArquivo(nomeArquivo, conteudoTexto, tipoLista) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    inicializarPlanilha();
    
    // Remove BOM (comum em arquivos exportados em UTF-8) antes de qualquer parsing
    conteudoTexto = String(conteudoTexto).replace(/^﻿/, '');

    var linhas = conteudoTexto.split(/\r?\n/);
    if (linhas.length === 0 || linhas[0].trim() === "") {
      return { success: false, error: "Arquivo vazio ou inválido." };
    }

    var primeiraLinha = linhas[0];

    // Detecta o delimitador contando qual separador mais aparece no cabeçalho
    var delimitador = detectarDelimitador(primeiraLinha);

    var cabecalho = primeiraLinha.split(delimitador).map(function(c) {
      return c.replace(/^["']|["']$/g, '').trim();
    });

    // Se não foi possível separar em colunas, o layout/delimitador é inválido
    if (cabecalho.length < 2) {
      var erroDelim = "Não foi possível identificar as colunas do arquivo. " +
        "Verifique se o separador é tabulação, ponto e vírgula (;) ou vírgula (,).";
      registrarUpload(nomeArquivo, tipoLista, 0, "Erro no Layout", erroDelim);
      return { success: false, error: erroDelim };
    }
    
    // Cabeçalhos Oficiais fornecidos pela Cresol
    var colunasEsperadas = [];
    if (tipoLista.indexOf("Cartão") !== -1) {
      colunasEsperadas = [
        "Agência", "Gerente", "Micro Segmento", "Documento", "Cooperado", "Conta Corrente", "Score", 
        "Ativo Comercial", "IRC", "Capital Social", "Saldo Devedor", "Aplicações", "DT Base SCR", 
        "Emprest. e Financ. SCR", "Emprest. e Financ. 360 SCR", "Emprest. e Financ. Mensal SCR", 
        "Renda Mensal", "Última Atualização Renda", "Compromet. Renda", "Renda Disponível", 
        "Cartão de Crédito - Limite SCR", "Cartão de Crédito - Limite Cresol", "Cartão de Crédito - Saldo Devedor", 
        "Cartão de Crédito - Sugestão Limite"
      ];
    } else {
      // Cheque Especial / Limite de Cheque
      colunasEsperadas = [
        "Agência", "Gerente", "Micro Segmento", "Documento", "Cooperado", "Conta Corrente", "Score", 
        "Ativo Comercial", "IRC", "Capital Social", "Saldo Devedor", "Aplicações", "DT Base SCR", 
        "Emprest. e Financ. SCR", "Emprest. e Financ. 360 SCR", "Emprest. e Financ. Mensal SCR", 
        "Renda Mensal", "Última Atualização Renda", "Compromet. Renda", "Renda Disponível", 
        "Cheque Especial - Limite SCR", "Cheque Especial - Limite Cresol", "Cheque Especial - Saldo Devedor", 
        "Cheque Especial - Sugestão Limite"
      ];
    }
    
    // Validar se colunas cruciais estão presentes
    // (comparação sem acentos para tolerar arquivos em ISO-8859-1/UTF-8)
    var colunasFaltantes = [];
    colunasEsperadas.forEach(function(col) {
      var colLimpa = normalizarCol(col);
      var achou = cabecalho.some(function(c) {
        return normalizarCol(c) === colLimpa;
      });
      if (!achou) {
        colunasFaltantes.push(col.replace(/\r?\n|\r/g, " "));
      }
    });
    
    if (colunasFaltantes.length > 0) {
      var erroMsg = "Layout incompatível. Faltando: " + colunasFaltantes.join(", ");
      registrarUpload(nomeArquivo, tipoLista, 0, "Erro no Layout", erroMsg);
      return { success: false, error: erroMsg, colunasFaltantes: colunasFaltantes };
    }
    
    // Mapear índices das colunas do arquivo para gravação
    var idx = {
      doc: -1, nome: -1, conta: -1, microSeg: -1, score: -1, ativoComercial: -1,
      irc: -1, capitalSocial: -1, dtBaseScr: -1, emprestScr: -1, emprest360: -1,
      emprestMensal: -1, renda: -1, comprometimento: -1, limiteScr: -1,
      saldoDevedor: -1, limiteCresol: -1, sugestao: -1, agencia: -1
    };

    cabecalho.forEach(function(c, i) {
      var cLimpo = normalizarCol(c);
      if (cLimpo === "documento") idx.doc = i;
      else if (cLimpo === "cooperado") idx.nome = i;
      else if (cLimpo === "conta corrente") idx.conta = i;
      else if (cLimpo === "micro segmento") idx.microSeg = i;
      else if (cLimpo === "score") idx.score = i;
      else if (cLimpo === "ativo comercial") idx.ativoComercial = i;
      else if (cLimpo === "irc") idx.irc = i;
      else if (cLimpo === "capital social") idx.capitalSocial = i;
      else if (cLimpo === "dt base scr") idx.dtBaseScr = i;
      else if (cLimpo === "emprest. e financ. scr") idx.emprestScr = i;
      else if (cLimpo === "emprest. e financ. 360 scr") idx.emprest360 = i;
      else if (cLimpo === "emprest. e financ. mensal scr") idx.emprestMensal = i;
      else if (cLimpo === "renda mensal") idx.renda = i;
      else if (cLimpo === "compromet. renda") idx.comprometimento = i;
      else if (cLimpo === "agencia") idx.agencia = i;
      else if (cLimpo.indexOf("limite cresol") !== -1) idx.limiteCresol = i;
      else if (cLimpo.indexOf("sugestao limite") !== -1) idx.sugestao = i;
      else if (cLimpo.indexOf("limite scr") !== -1) idx.limiteScr = i;
      else if (cLimpo.indexOf("saldo devedor") !== -1 &&
               (cLimpo.indexOf("cheque") !== -1 || cLimpo.indexOf("cartao") !== -1)) idx.saldoDevedor = i;
    });

    // Garante todas as colunas na aba Associados e obtém o mapa (0-based)
    var mapaA = garantirColunasAssociados();
    var shAssociados = ss.getSheetByName("Associados");
    var totalCols = shAssociados.getLastColumn();
    var dadosExistem = shAssociados.getDataRange().getValues();
    var colDoc = mapaA["cpf/cnpj"];
    var cpfsExistentes = {};
    for (var i = 1; i < dadosExistem.length; i++) {
      var cpf = String(dadosExistem[i][colDoc]).trim();
      cpfsExistentes[cpf] = i + 1;
    }
    
    var registrosProcessados = 0;
    
    for (var i = 1; i < linhas.length; i++) {
      var linhaTexto = linhas[i].trim();
      if (linhaTexto === "") continue;
      
      var colunas = linhaTexto.split(delimitador).map(function(c) {
        return c.replace(/^["']|["']$/g, '').trim();
      });

      if (colunas.length < cabecalho.length) continue;

      var pega = function(i) { return (i === -1) ? "" : (colunas[i] || ""); };

      var doc = pega(idx.doc);
      var nome = pega(idx.nome);
      if (!doc || !nome) continue;

      // Monta os campos derivados do arquivo (money via parseDinheiro; texto direto)
      var campos = {
        "nome": nome,
        "conta": pega(idx.conta),
        "limite atual": parseDinheiro(pega(idx.limiteCresol)),
        "sugestao motor": parseDinheiro(pega(idx.sugestao)),
        "status": "AGUARDANDO",
        "renda comprovada": parseDinheiro(pega(idx.renda)),
        "comprometimento scr": parseFloat(pega(idx.comprometimento)) || 0,
        "score": Number(pega(idx.score)) || 0,
        "micro segmento": String(pega(idx.microSeg)).trim(),
        "ativo comercial": String(pega(idx.ativoComercial)).trim(),
        "irc": String(pega(idx.irc)).trim(),
        "capital social": parseDinheiro(pega(idx.capitalSocial)),
        "dt base scr": String(pega(idx.dtBaseScr)).trim(),
        "emprest. e financ. scr": parseDinheiro(pega(idx.emprestScr)),
        "emprest. e financ. 360 scr": parseDinheiro(pega(idx.emprest360)),
        "emprest. e financ. mensal scr": parseDinheiro(pega(idx.emprestMensal)),
        "limite scr": parseDinheiro(pega(idx.limiteScr)),
        "saldo devedor": parseDinheiro(pega(idx.saldoDevedor))
      };
      var agencia = (idx.agencia !== -1) ? String(pega(idx.agencia)).trim() : "";
      if (agencia) campos["agencia"] = agencia;

      registrosProcessados++;
      var docFormatado = String(doc).trim();

      var setCampo = function(arr, nomeCol, valor) {
        var i = mapaA[nomeCol];
        if (i !== undefined) arr[i] = valor;
      };

      if (cpfsExistentes[docFormatado]) {
        // Atualiza: parte da linha existente e sobrescreve os campos do arquivo
        var l = cpfsExistentes[docFormatado];
        var linhaArr = dadosExistem[l - 1].slice();
        while (linhaArr.length < totalCols) linhaArr.push("");
        for (var nomeCol in campos) setCampo(linhaArr, nomeCol, campos[nomeCol]);
        shAssociados.getRange(l, 1, 1, totalCols).setValues([linhaArr]);
      } else {
        // Novo associado
        var novoId = "ASC-" + String(shAssociados.getLastRow() + 1).padStart(3, '0');
        var novaLinha = [];
        for (var k = 0; k < totalCols; k++) novaLinha.push("");
        setCampo(novaLinha, "id", novoId);
        setCampo(novaLinha, "cpf/cnpj", docFormatado);
        setCampo(novaLinha, "inadimplencia dias", 0);
        setCampo(novaLinha, "tempo associado (anos)", 3);
        for (var nomeCol2 in campos) setCampo(novaLinha, nomeCol2, campos[nomeCol2]);
        shAssociados.appendRow(novaLinha);
      }
    }
    
    registrarUpload(nomeArquivo, tipoLista, registrosProcessados, "Validado", "");
    return { success: true, registros: registrosProcessados };
  } catch (err) {
    registrarUpload(nomeArquivo, tipoLista, 0, "Erro no Layout", err.toString());
    return { success: false, error: err.toString() };
  }
}

// Identifica o colaborador logado (via conta Google) e sua agência na aba Colaboradores
function obterUsuarioAtual() {
  var resultado = { email: "", agencia: "", isSede: false, cadastrado: false };
  try {
    resultado.email = Session.getActiveUser().getEmail() || "";
  } catch (e) {
    resultado.email = "";
  }
  var emailNorm = String(resultado.email).trim().toLowerCase();
  if (!emailNorm) return resultado; // e-mail indisponível nesta implantação

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName("Colaboradores");
  if (!sh) return resultado;

  var dados = sh.getDataRange().getValues();
  for (var i = 1; i < dados.length; i++) {
    var e = String(dados[i][0] || "").trim().toLowerCase();
    if (e && e === emailNorm) {
      resultado.agencia = String(dados[i][2] || "").trim();
      resultado.cadastrado = true;
      resultado.isSede = normalizarCol(resultado.agencia).indexOf("sede") !== -1;
      break;
    }
  }
  return resultado;
}

// Monta um mapa {nomeColunaNormalizado: índice 0-based} a partir do cabeçalho da aba
function mapaColunas(sheet) {
  var lastCol = Math.max(sheet.getLastColumn(), 1);
  var header = sheet.getRange(1, 1, 1, lastCol).getValues()[0];
  var mapa = {};
  for (var i = 0; i < header.length; i++) {
    var n = normalizarCol(header[i]);
    if (n && !(n in mapa)) mapa[n] = i;
  }
  return mapa;
}

// Garante que a aba Associados possua todas as colunas canônicas (adiciona as
// faltantes ao final) e retorna o mapa {nomeNormalizado: índice 0-based}
function garantirColunasAssociados() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sh = ss.getSheetByName("Associados");
  if (!sh) return {};
  var lastCol = Math.max(sh.getLastColumn(), 1);
  var header = sh.getRange(1, 1, 1, lastCol).getValues()[0];
  var existentes = header.map(function(h) { return normalizarCol(h); });
  COLUNAS_ASSOCIADOS.forEach(function(col) {
    if (existentes.indexOf(normalizarCol(col)) === -1) {
      lastCol++;
      sh.getRange(1, lastCol).setValue(col);
      existentes.push(normalizarCol(col));
    }
  });
  return mapaColunas(sh);
}

// Detecta o delimitador do CSV/TSV escolhendo o separador mais frequente no cabeçalho
function detectarDelimitador(linha) {
  var candidatos = ["\t", ";", ",", "|"];
  var melhor = ";";
  var maxOcorrencias = -1;
  candidatos.forEach(function(d) {
    var ocorrencias = linha.split(d).length - 1;
    if (ocorrencias > maxOcorrencias) {
      maxOcorrencias = ocorrencias;
      melhor = d;
    }
  });
  return melhor;
}

// Normaliza um cabeçalho de coluna: remove quebras de linha, acentos e caixa
function normalizarCol(texto) {
  return String(texto)
    .replace(/\r?\n|\r/g, " ")
    .normalize('NFD').replace(/[̀-ͯ]/g, '')
    .trim()
    .toLowerCase();
}

function parseDinheiro(valorTexto) {
  if (!valorTexto) return 0;
  var limpo = String(valorTexto)
    .replace("R$", "")
    .replace(/\s/g, "")
    .replace(/\./g, "")
    .replace(",", ".");
  return Number(limpo) || 0;
}

// Helpers para Formatação de Data/Hora brasileira
function obterDataAtualFormatada() {
  var hoje = new Date();
  var dd = String(hoje.getDate()).padStart(2, '0');
  var mm = String(hoje.getMonth() + 1).padStart(2, '0');
  var yyyy = hoje.getFullYear();
  return dd + '/' + mm + '/' + yyyy;
}

function obterHoraAtualFormatada() {
  var hoje = new Date();
  var hh = String(hoje.getHours()).padStart(2, '0');
  var min = String(hoje.getMinutes()).padStart(2, '0');
  return hh + ':' + min;
}
